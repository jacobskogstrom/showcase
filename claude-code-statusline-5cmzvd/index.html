Left to right

What each part tells you

  • Opus 5.5 [high]
    Model and effort. I switch models during the day. This stops me running a big planning job on the small model, or burning the big one on a rename.
  • 37% contxt
    Context bar. How full the model's working memory is right now. When it climbs high, answers get vaguer and older instructions start to slip. That is my cue to compact, or write a handover and start fresh.
  • 3% sess.token
    Plan bar. How much of your subscription's 5-hour window you have used. Same number claude.ai shows under Settings > Usage, now without leaving the terminal. On API billing, or before the first reply, it falls back to tokens used vs the context window.
  • "save this..."
    Last prompt. The first 40 characters of what you last typed. With five terminals open, this is how you know which tab is which.

The two bars are the point. Context is what silently degrades your results. Plan usage is what silently stops you mid-task.

Two minutes

Install it

You need Node.js 18 or newer. No packages, nothing leaves your machine. The script reads the JSON Claude Code pipes into every status line command, plus your own session transcript for the last prompt.

  1. Save the script below as ~/.claude/statusline.js. On Windows that is C:\Users\<you>\.claude\statusline.js.

    statusline.js

  2. Add this to ~/.claude/settings.json, then restart Claude Code. On Windows use the full path with forward slashes: node C:/Users/<you>/.claude/statusline.js.

    settings.json
    "statusLine": {
  "type": "command",
  "command": "node ~/.claude/statusline.js",
  "padding": 0
}

The script

statusline.js
// statusline.js - a Claude Code status line that answers three questions at a glance:
//   which model and effort am I on, how full is my context, how much of my plan window is used.
//
// Renders:  Opus 5.5 [high] [████░░░░░░]37% contxt [░░░░░░░░░░]3% sess.token, "your last prompt..."
//
// Install: save this file as ~/.claude/statusline.js, then add to ~/.claude/settings.json:
//   "statusLine": { "type": "command", "command": "node ~/.claude/statusline.js", "padding": 0 }
// (Windows: use the full path with forward slashes, e.g. "node C:/Users/<you>/.claude/statusline.js")
// Needs Node.js 18+. No dependencies. Reads only what Claude Code pipes in plus your own transcript.
//
// Free to share. From Telling Technology: https://tellingtechnology.skogstrom.se

'use strict';
const fs = require('fs');

const ESC = '\x1b[';
const paint = (code) => (s) => ESC + code + 'm' + s + ESC + '0m';
const dim = paint('90'), cyan = paint('36'), yellow = paint('33');

function bar(pct) {
  const width = 10;
  const fill = Math.round((pct / 100) * width);
  return '[' + cyan('█'.repeat(fill)) + dim('░'.repeat(width - fill)) + ']';
}

function clampPct(n) {
  return Math.max(0, Math.min(100, Math.round(Number(n) || 0)));
}

function readStdin() {
  const chunks = [];
  const buf = Buffer.alloc(65536);
  for (;;) {
    let n = 0;
    try { n = fs.readSync(0, buf, 0, buf.length, null); } catch { break; }
    if (n <= 0) break;
    chunks.push(buf.slice(0, n).toString('utf8'));
  }
  const raw = chunks.join('').replace(/^\uFEFF/, '').trim();
  return raw ? JSON.parse(raw) : {};
}

function lastUserPrompt(transcriptPath) {
  if (!transcriptPath || !fs.existsSync(transcriptPath)) return '';
  const lines = fs.readFileSync(transcriptPath, 'utf8').split('\n').filter(Boolean).slice(-200);
  for (let i = lines.length - 1; i >= 0; i--) {
    let entry;
    try { entry = JSON.parse(lines[i]); } catch { continue; }
    if (entry.type !== 'user' && entry.role !== 'user') continue;
    const content = (entry.message || entry).content;
    let text = '';
    if (typeof content === 'string') text = content;
    else if (Array.isArray(content)) {
      const block = content.find((b) => b && b.type === 'text' && b.text && !b.text.trim().startsWith('<'));
      text = block ? block.text : '';
    }
    text = text.replace(/\s+/g, ' ').trim();
    if (text && !text.startsWith('<')) return text;
  }
  return '';
}

let data = {};
try { data = readStdin(); } catch { /* render the fallback */ }

const model = String((data.model && (data.model.display_name || data.model.id)) || 'Claude')
  .replace(/^Claude\s+/i, '').replace(/\s*\(.*\)\s*$/, '').trim() || 'Claude';
const effort = data.effort && data.effort.level;

const cw = data.context_window || {};
const size = Number(cw.context_window_size) || 200000;
const inTok = Number(cw.total_input_tokens) || 0;
const outTok = Number(cw.total_output_tokens) || 0;
const contextPct = clampPct(cw.used_percentage != null ? cw.used_percentage : (inTok / size) * 100);

// Plan usage: the same "Current session X% used" number claude.ai shows under Settings > Usage
// (Pro/Max, rolling 5-hour window). Before the first reply, or on API billing, fall back to tokens used vs window.
const fiveHour = data.rate_limits && data.rate_limits.five_hour && data.rate_limits.five_hour.used_percentage;
const sessionPct = clampPct(fiveHour != null ? fiveHour : ((inTok + outTok) / size) * 100);

let prompt = '';
try { prompt = lastUserPrompt(data.transcript_path); } catch { /* omit */ }
if (prompt.length > 40) prompt = prompt.slice(0, 40) + '...';

const parts = [yellow(model)];
if (effort) parts.push('[' + effort + ']');
parts.push(bar(contextPct) + contextPct + '% contxt');
parts.push(bar(sessionPct) + sessionPct + '% sess.token');
process.stdout.write(parts.join(' ') + (prompt ? ', ' + dim('"' + prompt + '"') : '') + '\n');

Make it yours

  • Change 40 to show more of the last prompt.
  • Change width = 10 for longer bars.
  • Colors are plain ANSI codes at the top: 33 yellow, 36 cyan, 90 grey.
  • Want cost, git branch or folder name? The same JSON carries cost, workspace.current_dir and more. Ask Claude Code to add a segment; it can read the payload itself.
